A2. sampleJitterNoise / sampleIGNJitter

inline float sampleJitterNoise(float2 st, float viewportH, float blockSize,
    texture2d<float> noiseTexture, float useTextureJitter) {
  if (useTextureJitter > 0.5f)
  {
    // Correlated noise-texture sampling (production default, 2026-08-19):
    // sample the 64x64 blue-noise tile with LINEAR + REPEAT at position.xy/64,
    // y-flipped for Metal's flipped texture origin vs GL. Bilinear filtering
    // correlates the jitter field spatially so adjacent SIMT lanes stay near
    // lockstep — jitter_lag_repro measured +24% j1 at 2048/SD4 vs +58% for
    // per-pixel point sampling (the legacy path below), closing the M/GL gap
    // from 1.33x to 1.04x. The filter also lowers the tile's high-frequency
    // amplitude (anti-banding) instead of quantizing it to viewport/64 px
    // blocks like GL's NEAREST sampling.
    constexpr sampler noiseSampler(filter::linear, address::repeat);
    return noiseTexture.sample(noiseSampler, float2(st.x, viewportH - st.y) / 64.0f).r;
  }
  // GL-parity per-pixel sampling (2026-08-19): GL samples the 64x64 tile at
  // gl_FragCoord.xy/64 with NEAREST + REPEAT, i.e. texel = pixel mod 64 —
  // PER-PIXEL (the tile repeats every 64px), which is why GL's jitter is fine
  // grain, not 32px blocks. The earlier block-division form (floor(x/texel))
  // produced visible 32px ripple blocks and was rejected. The correct GL
  // field: kBlueNoise64[(pixel.y mod 64) * 64 + (pixel.x mod 64)] with GL's
  // bottom-up y.
  if (blockSize < 0.5f)
  {
    int2 t = int2(int(st.x) & 63, int(viewportH - st.y) & 63);
    return float(kBlueNoise64[t.y * 64 + t.x]) / 255.0f;
  }
  // Block-coherent sampling (2026-08-18): quantize the pixel coordinate to
  // the jitter block cell so adjacent pixels march in lockstep (per-pixel
  // jitter scatters warps and costs ~+57% on the raw path; block4 brings it
  // to GL's +27% level). blockSize 1 = legacy per-pixel (bit-identical).
  // NOTE: never divide by the raw blockSize — nSize 0 (GL-parity above)
  // would make this div-by-zero NaN and trap the shader under fast-math
  // speculation (the black-render bug, 2026-08-19).
  float bs = max(blockSize, 1.0f);
  st = floor(st / bs) * bs + 0.5f * bs;
  int2 t = int2(floor(st.x), floor(st.y - viewportH)) & 63;
  return float(kBlueNoise64[t.y * 64 + t.x]) / 255.0f;
}

// Interleaved Gradient Noise (Jimenez 2014), the jitter used before the GL
// blue-noise tile landed. Smooth, low-discrepancy, deterministic per pixel,
// no sin-hash streaks, no texture required. Selected per-render via
// volumeUniforms.useIGNJitter; the default stays on the GL-parity blue noise.
inline float sampleIGNJitter(float2 st, float blockSize) {
  // Never divide by the raw blockSize: nSize 0 (GL-parity) would make this
  // div-by-zero NaN (fast-math) and kill the march (2026-08-19).
  blockSize = max(blockSize, 1.0f);
  float2 blockCenter = floor(st / blockSize) * blockSize + 0.5f * blockSize;
  return fract(52.9829189 * fract(dot(blockCenter, float2(0.06711056, 0.00583715))));
}

inline float sampleVolumeScalar(texture3d<float> volTex, float3 pos) {
  if (fc_marchVariant == 1)  // manual 8-tap trilinear (see above)
  {
      return volTex.sample(sVolumeClampZero, pos, level(0)).r;
  }

int vtkOpenGLRenderWindow::GetNoiseTextureUnit()
{
  if (!this->NoiseTextureObject)
  {
    this->NoiseTextureObject = vtkTextureObject::New();
    this->NoiseTextureObject->SetContext(this);
  }

  if (this->NoiseTextureObject->GetHandle() == 0)
  {
    vtkNew<vtkJPEGReader> textureReader;
    textureReader->SetFileName("BlueNoiseTexture64x64.jpg");
    textureReader->Update();

    vtkImageData* image = textureReader->GetOutput();
    int* dims = image->GetDimensions();
    int bufferSize = dims[0] * dims[1];
    float* noiseTextureData = new float[bufferSize];

    for (int i = 0; i < bufferSize; i++)
    {
      noiseTextureData[i] = textureReader->GetScalarComponentAsFloat(0, 0, 0, 0) / 255.0f;
    }

    this->NoiseTextureObject->Create2DFromRaw(64, 64, 1, VTK_FLOAT, noiseTextureData);
    this->NoiseTextureObject->SetWrapS(vtkTextureObject::Repeat);
    this->NoiseTextureObject->SetWrapT(vtkTextureObject::Repeat);
    this->NoiseTextureObject->SetMagnificationFilter(vtkTextureObject::Nearest);
    this->NoiseTextureObject->SetMinificationFilter(vtkTextureObject::Nearest);
    delete[] noiseTextureData;
  }

  int result = this->GetTextureUnitForTexture(this->NoiseTextureObject);
  this->NoiseTextureObject->Activate();
  return this->GetTextureUnitForTexture(this->NoiseTextureObject);
}

### 2.4 Divergence isolation (probe)

The real shader's loop (`marchVolumeUnified`) breaks at a per-fragment,
data-dependent iteration (per-fragment `maxSteps`, CTP-bounds exit,
`tTerminateMax`, opacity threshold). A probe variant that runs a **fixed,
uniform** iteration count for all fragments (no break) drops the march from
~103 ms (divergent) to ~72.7 ms (fixed). A variant that keeps the uniform count
but uses predication (`select`) instead of a break - so the result is
image-correct and the fetch stays unconditional - measures **72.2 ms**.
